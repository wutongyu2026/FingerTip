```rust
//! v0.3: generate 层抽象。所有真正的生成实现都走 `MusicAdapter` / `ArtAdapter` trait，
//! 默认 `LocalXxxAdapter`（本地算法），cloud 仅当 `FINGERTIP_USE_CLOUD=1` 环境变量启用。
//!
//! 历史：
//!   - v0.2.x: `MinimaxCloudAdapter` 是 stub 不联网，已在 v0.3 删除
//!   - v0.3+: `LocalMusicAdapter` / `LocalArtAdapter` 是真生成（mood/theme 真影响）

pub mod local;
pub mod cloud;

use crate::hook::event::KeyEvent;
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone)]
pub struct MusicPrompt {
    pub events: Vec<KeyEvent>,
    pub mood: Option<String>,
    pub style: String,
    pub theme_word: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct NoteSpec {
    pub key_code: u32,
    pub velocity: u8,
    pub start_ms: u64,
    pub duration_ms: u64,
}

#[derive(Debug, Clone, Serialize)]
pub struct Music {
    pub notes: Vec<NoteSpec>,
    pub bpm: u32,
    pub duration_ms: u64,
    pub amplitudes: Vec<f32>,   // 64 采样点；前端用于渲染 36~64 bar 波形
    pub mood: Option<String>,
    pub style: String,
    pub theme_word: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct Art {
    pub width: u32,
    pub height: u32,
    pub pixels: Vec<PixelSpec>,
    pub palette_seed: u32,
    pub theme_word: String,
    pub mood: Option<String>,
}

#[derive(Debug, Clone, Serialize)]
pub struct PixelSpec {
    pub x: u32,
    pub y: u32,
    pub rgba: [u8; 4],
}

#[async_trait::async_trait]
pub trait MusicAdapter: Send + Sync {
    fn name(&self) -> &'static str;
    async fn generate(&self, prompt: &MusicPrompt) -> anyhow::Result<Music>;
}

#[async_trait::async_trait]
pub trait ArtAdapter: Send + Sync {
    fn name(&self) -> &'static str;
    async fn generate(&self, prompt: &MusicPrompt) -> anyhow::Result<Art>;
}

/// Factory: 选 Local 或 Cloud，默认 Local
pub fn build_music_adapter() -> Box<dyn MusicAdapter> {
    if std::env::var("FINGERTIP_USE_CLOUD").is_ok() {
        Box::new(cloud::music::CloudMusicAdapter)
    } else {
        Box::new(local::music::LocalMusicAdapter)
    }
}

pub fn build_art_adapter() -> Box<dyn ArtAdapter> {
    if std::env::var("FINGERTIP_USE_CLOUD").is_ok() {
        Box::new(cloud::art::CloudArtAdapter)
    } else {
        Box::new(local::art::LocalArtAdapter)
    }
}
```

```rust
#[tauri::command]
pub async fn generate_now_impl(
    state: State<'_, AppState>,
    date: String,
    mood: Option<String>,
    style: String,
) -> Result<String, String> {
    let conn = state.conn.lock().map_err(|e| e.to_string())?;
    let events = crate::db::event_repo::EventRepo::new(&conn)
        .list_by_date(&date)
        .map_err(|e| e.to_string())?;
    let summary = crate::db::summary_repo::SummaryRepo::new(&conn)
        .read_by_date(&date)
        .map_err(|e| e.to_string())?
        .ok_or_else(|| "today_summary missing; run key capture first".to_string())?;
    
    let prompt = crate::generate::MusicPrompt {
        events,
        mood,
        style,
        theme_word: summary.theme_word.unwrap_or_default(),
    };
    
    let music_adapter = crate::generate::build_music_adapter();
    let art_adapter = crate::generate::build_art_adapter();
    let music = music_adapter.generate(&prompt).await
        .map_err(|e| format!("{}: {}", music_adapter.name(), e))?;
    let art = art_adapter.generate(&prompt).await
        .map_err(|e| format!("{}: {}", art_adapter.name(), e))?;
    
    Ok(serde_json::json!({
        "music": music,
        "art": art,
    }).to_string())
}
```

```rust
pub fn list_by_date(&self, date: &str) -> anyhow::Result<Vec<KeyEvent>> {
    // 用 today 的 0:00~24:00 UTC 范围过滤（与 get_today_hourly 同假设）
    let (start, end) = crate::commands::tz_today_range_ms(0);
    self.0.prepare(
        "SELECT key_code, is_press, timestamp_ms, session_id FROM key_events
         WHERE timestamp_ms >= ?1 AND timestamp_ms < ?2 ORDER BY timestamp_ms ASC",
    )?
    .query_map(rusqlite::params![start, end], |row| {
        Ok(KeyEvent {
            key_code: row.get::<_, u32>(0)?,
            is_press: row.get::<_, bool>(1)?,
            timestamp_ms: row.get::<_, i64>(2)?,
            session_id: row.get::<_, String>(3)?,
        })
    })?
    .collect::<Result<Vec<_>, _>>()
    .map_err(Into::into)
}
```

```rust
use crate::generate::{MusicAdapter, MusicPrompt, Music, NoteSpec};

pub struct LocalMusicAdapter;

#[async_trait::async_trait]
impl MusicAdapter for LocalMusicAdapter {
    fn name(&self) -> &'static str { "local" }

    async fn generate(&self, prompt: &MusicPrompt) -> anyhow::Result<Music> {
        let bpm = mood_to_bpm(prompt.mood.as_deref().unwrap_or("neutral"));
        let duration_ms = compute_duration_ms(&prompt.events, bpm);
        let notes = events_to_notes(&prompt.events, bpm);
        let amplitudes = compute_amplitudes(&prompt.events, &prompt.theme_word);
        Ok(Music {
            notes,
            bpm,
            duration_ms,
            amplitudes,
            mood: prompt.mood.clone(),
            style: prompt.style.clone(),
            theme_word: prompt.theme_word.clone(),
        })
    }
}

fn mood_to_bpm(mood: &str) -> u32 {
    match mood {
        "calm" | "sad" | "reflective" | "tired" => 80,
        "happy" | "excited" | "energetic" | "focused" => 130,
        "angry" | "frustrated" => 150,
        _ => 100, // neutral / unknown
    }
}

fn compute_duration_ms(events: &[KeyEvent], bpm: u32) -> u64 {
    if events.is_empty() { return (60_000 / bpm as u64) * 16; } // 默认 16 拍
    let first = events.first().unwrap().timestamp_ms;
    let last = events.last().unwrap().timestamp_ms;
    ((last - first).max(0) as u64).max(8_000) // 至少 8s
}

fn events_to_notes(events: &[KeyEvent], bpm: u32) -> Vec<NoteSpec> {
    if events.is_empty() { return vec![]; }
    let base_ms = events.first().unwrap().timestamp_ms;
    let beat_ms = 60_000 / bpm as u64;
    events.iter().take(64).enumerate().map(|(i, e)| NoteSpec {
        key_code: e.key_code,
        velocity: 80,
        start_ms: (e.timestamp_ms - base_ms) as u64,
        duration_ms: beat_ms / 2,
    }).collect()
}

fn compute_amplitudes(events: &[KeyEvent], theme_word: &str) -> Vec<f32> {
    let mut amps = vec![0.0f32; 64];
    if events.is_empty() {
        // fallback: theme_word 字符 sum 当作 amplitude
        let sum: u32 = theme_word.chars().map(|c| c as u32).sum();
        for a in amps.iter_mut() {
            *a = ((sum % 100) as f32) / 100.0;
        }
        return amps;
    }
    // 把 events 按 64 个桶分
    let bucket_count = 64usize;
    let first = events.first().unwrap().timestamp_ms;
    let last = events.last().unwrap().timestamp_ms;
    let span = (last - first).max(1) as f64;
    let mut counts = vec![0u32; bucket_count];
    for e in events {
        let bucket = (((e.timestamp_ms - first) as f64 / span) * (bucket_count - 1) as f64) as usize;
        counts[bucket.min(bucket_count - 1)] += 1;
    }
    // theme_word hash 作为放大系数
    let theme_factor: f32 = (theme_word.bytes().map(|b| b as u32).sum::<u32>() % 50) as f32 / 100.0 + 0.75;
    let max_count = *counts.iter().max().unwrap_or(&1).max(&1) as f32;
    for (i, c) in counts.iter().enumerate() {
        amps[i] = ((*c as f32) / max_count) * theme_factor;
    }
    amps
}
```

```rust
use crate::generate::{Art, ArtAdapter, MusicPrompt, PixelSpec};

pub struct LocalArtAdapter;

#[async_trait::async_trait]
impl ArtAdapter for LocalArtAdapter {
    fn name(&self) -> &'static str { "local" }

    async fn generate(&self, prompt: &MusicPrompt) -> anyhow::Result<Art> {
        let width = 256u32;
        let height = 256u32;
        let pixels = compute_pixels(&prompt.events, &prompt.theme_word, prompt.mood.as_deref().unwrap_or("neutral"), width, height);
        Ok(Art {
            width, height,
            pixels,
            palette_seed: theme_word_seed(&prompt.theme_word),
            theme_word: prompt.theme_word.clone(),
            mood: prompt.mood.clone(),
        })
    }
}

fn theme_word_seed(tw: &str) -> u32 {
    tw.bytes().fold(0u32, |acc, b| acc.wrapping_mul(31).wrapping_add(b as u32))
}

fn mood_base_hue(mood: &str) -> u16 {
    match mood {
        "happy" | "energetic" | "excited" => 30,    // 暖橙红
        "calm" | "neutral" => 120,                  // 中性绿
        "sad" | "tired" => 200,                     // 冷蓝
        "angry" => 0,                               // 红
        "reflective" => 250,                        // 紫
        _ => 90,                                    // 默认橄榄
    }
}

fn compute_pixels(events: &[crate::hook::event::KeyEvent], theme_word: &str, mood: &str, width: u32, height: u32) -> Vec<PixelSpec> {
    let n = 64usize;
    let base_hue = mood_base_hue(mood);
    let theme_seed = theme_word_seed(theme_word);
    (0..n).map(|i| {
        // hue 由 base + 字符驱动
        let char_hue = theme_word.bytes().nth(i % theme_word.len().max(1)).unwrap_or(b'x') as u16;
        let h = (base_hue as u32 + char_hue as u32) % 360;
        let (r, g, b) = hsv_to_rgb(h as f32, 0.7, 0.85);
        // 坐标由 key_code hash + theme_seed 驱动
        let x = ((theme_seed.wrapping_add(i as u32 * 97)) % width) as u32;
        let y = if !events.is_empty() {
            let ev = &events[i % events.len()];
            ((ev.key_code as u32 * 11 + i as u32 * 13) % height) as u32
        } else {
            ((i as u32 * 19 + theme_seed) % height) as u32
        };
        PixelSpec { x, y, rgba: [r, g, b, 255] }
    }).collect()
}

fn hsv_to_rgb(h: f32, s: f32, v: f32) -> (u8, u8, u8) {
    let h = h % 360.0;
    let c = v * s;
    let x = c * (1.0 - ((h / 60.0) % 2.0 - 1.0).abs());
    let m = v - c;
    let (r1, g1, b1) = if h < 60.0 { (c, x, 0.0) }
        else if h < 120.0 { (x, c, 0.0) }
        else if h < 180.0 { (0.0, c, x) }
        else if h < 240.0 { (0.0, x, c) }
        else if h < 300.0 { (x, 0.0, c) }
        else { (c, 0.0, x) };
    (((r1 + m) * 255.0) as u8, ((g1 + m) * 255.0) as u8, ((b1 + m) * 255.0) as u8)
}
```

### Task 5.2: 创建 download 工具函数

**Files:**
- Create: `src/utils/download.ts`

**Step 1: 写文件**

```typescript
import { save } from '@tauri-apps/plugin-dialog'
import { writeFile } from '@tauri-apps/plugin-fs'
import { useAppStore } from '@/stores/app'

export async function downloadBlob(filename: string, blob: Blob, ext: string): Promise<void> {
  const dir = useAppStore().downloadDir || await ensureDefaultDir()
  const defaultName = filename.endsWith(`.${ext}`) ? filename : `${filename}.${ext}`
  const path = await save({
    defaultPath: `${dir}/${defaultName}`,
    filters: [{ name: ext.toUpperCase(), extensions: [ext] }],
  })
  if (!path) return  // user cancelled
  const bytes = new Uint8Array(await blob.arrayBuffer())
  await writeFile(path, bytes)
}

export async function ensureDefaultDir(): Promise<string> {
  const store = useAppStore()
  if (store.downloadDir) return store.downloadDir
  const { appDir, join } = await import('@tauri-apps/api/path')
  const dir = await join(await appDir(), 'downloads')
  await writeFile(dir + '/.init', new Uint8Array())  // ensure exists
  store.setDownloadDir(dir)
  return dir
}
```
